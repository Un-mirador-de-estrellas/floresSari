function typeName(name) { title.innerHTML = ''; name.split('').forEach((l, i) => { const span = document.createElement('span'); span.textContent = l; span.style.opacity = 0; span.style.animation = `appear 0.5s forwards ${i*0.3}s`; title.appendChild(span); }); } function typeLetter(text) { letterText.textContent = ''; let i = 0; const interval = setInterval(() => { letterText.textContent += text[i]; i++; if (i >= text.length) clearInterval(interval); }, 40); } function createFalling() { const items = ['🌼','💖','🐱']; const el = document.createElement('div'); el.classList.add('fall'); el.innerHTML = items[Math.floor(Math.random()*items.length)]; el.style.left = Math.random()*window.innerWidth + 'px'; el.style.animationDuration = (3 + Math.random()*5)+'s'; document.body.appendChild(el); setTimeout(()=>el.remove(),8000); } function createSpark(x, y) { for (let i = 0; i < 8; i++) { // Reducido un poco para mejor rendimiento const s = document.createElement('div'); s.classList.add('spark'); s.style.left = x + 'px'; s.style.top = y + 'px'; s.style.setProperty('--x', (Math.random()-0.5)*100 + 'px'); s.style.setProperty('--y', (Math.random()-0.5)*100 + 'px'); document.body.appendChild(s); setTimeout(()=>s.remove(),1000); } } function createFirework(x, y) { for (let i = 0; i < 20; i++) { const fw = document.createElement('div'); fw.classList.add('firework'); fw.style.left = x + 'px'; fw.style.top = y + 'px'; fw.style.setProperty('--x', (Math.random()-0.5)*300 + 'px'); fw.style.setProperty('--y', (Math.random()-0.5)*300 + 'px'); document.body.appendChild(fw); setTimeout(()=>fw.remove(),1000); } } function createBouquet(x, y, scale=1) { for (let i = 0; i < 12; i++) { const flower = document.createElement('div'); flower.classList.add('flower-head'); flower.innerHTML = '🌼'; const offsetX = (Math.random() - 0.5) * 60 * scale; const offsetY = Math.random() * 60 * scale; flower.style.left = (x + offsetX) + 'px'; flower.style.top = (y - offsetY) + 'px'; flower.style.fontSize = (30 * scale) + 'px'; document.body.appendChild(flower); setTimeout(()=> flower.style.transform = 'scale(1)', 100); } // ¡CORRECCIÓN CLAVE!: Movemos las chispas fuera del bucle de las 12 flores. // Ahora solo se crean una vez por ramo, no 12 veces por ramo. createSpark(x, y); } function createPlant(x, origin, scale=1) { const plant = document.createElement('div'); plant.classList.add('plant'); const size = (100 + Math.random()*80) * scale; let endX = x; let endY = window.innerHeight; if (origin === 'bottom') { plant.style.bottom = 0; plant.style.left = x + 'px'; setTimeout(()=> plant.style.height = size + 'px', 50); endY = window.innerHeight - size; } if (origin === 'top') { plant.style.top = 0; plant.style.left = x + 'px'; setTimeout(()=> plant.style.height = size + 'px', 50); endY = size; } if (origin === 'left') { plant.style.left = 0; plant.style.top = x + 'px'; plant.style.height = '4px'; setTimeout(()=> plant.style.width = size + 'px', 50); endX = size; endY = x; } if (origin === 'right') { plant.style.right = 0; plant.style.top = x + 'px'; plant.style.height = '4px'; setTimeout(()=> plant.style.width = size + 'px', 50); endX = window.innerWidth - size; endY = x; } document.body.appendChild(plant); setTimeout(()=>{ createBouquet(endX, endY, scale); }, 2000); } function closePopup() { popup.style.display = 'none'; } btn.onclick = () => { btn.style.display = 'none'; typeName('Para Sari 💛'); // Reproducir música (asegúrate de que "prueba.mp3" esté en la misma carpeta) music.play().catch(e => console.log("El navegador bloqueó el audio temporalmente")); const origin = origins[Math.floor(Math.random()*origins.length)]; const scale = Math.random()*0.8 + 0.6; createPlant(x, origin, scale); }, i*120); // Aumenté un poquito el tiempo entre flor y flor (120ms) para darle respiro al navegador } let pos = 0;
